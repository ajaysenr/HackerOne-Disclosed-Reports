## Vulnerability Details

## Summary:

(Note: AI created the PoC, not the report.)

In the `krb5_read_data()` function [here](https://github.com/curl/curl/blob/af7d67d3c03329116e593d999851d2cc3ebbf119/lib/krb5.c#L526-L551), there are two issues (one of which I am very surprised hasn't been caught before.)

Issue #1 is that [this block](https://github.com/curl/curl/blob/af7d67d3c03329116e593d999851d2cc3ebbf119/lib/krb5.c#L539C1-L541C30) can result in an int overflow, where the following check becomes invalid if the value is wrapped to a negative:

```
    len = (int)ntohl((uint32_t)len);
    if(len > CURL_MAX_INPUT_LENGTH)
      return CURLE_TOO_LARGE;
```

This will then seemingly completely mess up the logic [here](https://github.com/curl/curl/blob/af7d67d3c03329116e593d999851d2cc3ebbf119/lib/krb5.c#L549-L551):

```
    char buffer[1024];
    nread = CURLMIN(len, (int)sizeof(buffer));
    result = socket_read(data, sockindex, buffer, (size_t)nread);
```

Since -1 (let's say is less than 1024, `nread` will be passed as `(size_t)-1`, which will wrap to a massive number.

This eventually makes its way down to `Curl_conn_recv` with `data[1024]` and `nread == ~SIZE_MAX`) [here](https://github.com/curl/curl/blob/af7d67d3c03329116e593d999851d2cc3ebbf119/lib/krb5.c#L486) and so on and so on.

It seems that this will result in an OOB read, as the buffer is only 1024-bytes long, while `nread` is going to be massive.

Issue #2 is that this whole code seems broken. [This do-while](https://github.com/curl/curl/blob/af7d67d3c03329116e593d999851d2cc3ebbf119/lib/krb5.c#L558) loop continues while `len` is non-zero. But then `len=0` is passed to `decode()` [here](https://github.com/curl/curl/blob/af7d67d3c03329116e593d999851d2cc3ebbf119/lib/krb5.c#L560-L562). As a random guess, I imagine `curlx_dyn_len(&buf->buf)` should be passed, not `len`.

static int socket_read_full(int fd, void *buf, size_t want) {
  uint8_t *p = buf; size_t got = 0;
  while (got < want) {
    ssize_t r = read(fd, p + got, want - got);
    if (r < 0) { if (errno == EINTR) continue; perror("read"); return -1; }
    if (r == 0) { fprintf(stderr, "eof\n"); return -1; }
    got += (size_t)r;
  }
  return 0;
}

  fprintf(stderr, "fake_decode() was called with len=%d (should be >0 if correct)\n", len);

  return 1234;

  if (len) {
    // network to host, then cast back to int (same as original)
    len = (int)ntohl((uint32_t)len);

    // Size check done against 'int' (problematic if sign bit set)

      fprintf(stderr, "too large\n");

  } else {

  size_t cap = 0;
  // Read loop that decrements len to zero

    int nread = len < 1024 ? len : 1024; // CURLMIN(len, sizeof buffer)
    dyn = realloc(dyn, cap + (size_t)nread);
    if (!dyn) { perror("realloc"); return 1; }
    if (socket_read_full(fd, dyn + cap, (size_t)nread) != 0) return 1;
    cap += (size_t)nread;
    len -= nread;
  }

  // At this point 'len' is 0, but 'cap' is the actual number of bytes we read.
  fprintf(stderr, "after loop: actual_buffer_len=%zu, len_variable=%d\n", cap, len);

  int nread = fake_decode(NULL, dyn, len, /*data_prot*/0, NULL); // <-- passes 0
  if (nread < 0) { fprintf(stderr, "decode error\n"); return 1; }

  fprintf(stderr, "fake_decode returned %d; we would then set dynbuf length to that.\n", nread);
  free(dyn);
  close(fd);
  return 0;

typedef enum {

  CURLE_TOO_LARGE = 1,
  CURLE_RECV_ERROR = 2,
  CURLE_AGAIN = 3,
  CURLE_FAILED_INIT = 4,

#define CURL_MAX_INPUT_LENGTH (16 * 1024 * 1024) // 16 MiB for demo
#define CURLMIN(a,b) ((a) < (b) ? (a) : (b))

struct dynbuf {
  char *ptr;
  size_t len;
  size_t cap;
};

static void dyn_init(struct dynbuf *d) { d->ptr = NULL; d->len = d->cap = 0; }
static void dyn_reset(struct dynbuf *d) { d->len = 0; }
static int dyn_grow(struct dynbuf *d, size_t need) {
  if (d->cap - d->len >= need) return 0;
  size_t newcap = d->cap ? d->cap : 1024;
  while (newcap - d->len < need) {
    newcap *= 2;
  }
  char *np = (char*)realloc(d->ptr, newcap);
  if (!np) return -1;
  d->ptr = np; d->cap = newcap;
  return 0;
}
static CURLcode dyn_addn(struct dynbuf *d, const char *src, size_t n) {
  if (dyn_grow(d, n) != 0) return CURLE_RECV_ERROR;
  memcpy(d->ptr + d->len, src, n);
  d->len += n;
  return CURLE_OK;
}
static char* dyn_ptr(struct dynbuf *d) { return d->ptr; }
static size_t dyn_len(struct dynbuf *d) { return d->len; }
static void dyn_setlen(struct dynbuf *d, size_t n) { if (n <= d->cap) d->len = n; }

static CURLcode Curl_conn_recv(struct Curl_easy *data, int sockindex,
                               char *buf, size_t blen, size_t *pnread) {
  (void)data;
  int fd = data->conn->fd[sockindex];
  ssize_t r = recv(fd, buf, blen, 0); // passes 'blen' straight to kernel
  if (r < 0) {
    if (errno == EAGAIN || errno == EWOULDBLOCK) return CURLE_AGAIN;
    return CURLE_RECV_ERROR;
  }
  *pnread = (size_t)r;
  return CURLE_OK;
}

static CURLcode socket_read(struct Curl_easy *data, int sockindex, void *to, size_t len) {
  char *to_p = (char*)to;
  CURLcode result;
  size_t nread = 0;
  while (len > 0) {
    result = Curl_conn_recv(data, sockindex, to_p, len, &nread);
    if (result == CURLE_AGAIN) continue;
    if (result) return result;
    if (nread > len) return CURLE_RECV_ERROR;
    len -= nread;
    to_p += nread;
  }
  return CURLE_OK;
}

// --- The vulnerable function as in the snippet (bug preserved) ---
static CURLcode krb5_read_data_vuln(struct Curl_easy *data, int sockindex,
                                    struct krb5buffer *kbuf) {
  int len; // <-- signed
  CURLcode result;
  int nread;

  result = socket_read(data, sockindex, &len, sizeof(len));
  if (result) return result;

  if (len) {
    len = (int)ntohl((uint32_t)len); // potential negative value
    if (len > CURL_MAX_INPUT_LENGTH) // negative bypasses this check
      return CURLE_TOO_LARGE;
    dyn_reset(&kbuf->buf);
  } else {
    return CURLE_RECV_ERROR;
  }
  do {
    char buffer[1024]; // fixed stack buffer
    nread = CURLMIN(len, (int)sizeof(buffer)); // if len<0 -> nread<0
    result = socket_read(data, sockindex, buffer, (size_t)nread); // cast to huge size_t
    if (result) return result;
    result = dyn_addn(&kbuf->buf, buffer, (size_t)nread);
    if (result) return result;
    len -= nread;
  } while (len);

  // pretend-decode, but len is now 0 in this code path
  int dec = (int)dyn_len(&kbuf->buf);
  if (dec < 0) return CURLE_RECV_ERROR;
  dyn_setlen(&kbuf->buf, (size_t)dec);
  kbuf->index = 0;
  return CURLE_OK;
}

struct writer_args { int fd; int mode; };

static void* writer_thread(void *argp) {

When building this and running it as `./a.out poc`, it does not cause any problem:

```
$ ./a overflow
krb5_read_data_vuln returned code: 2
```

So what's going on? The "problem" is this call:

```
  int fd = data->conn->fd[sockindex];
  ssize_t r = recv(fd, buf, blen, 0); // passes 'blen' straight to kernel
```
With `blen` now effectively `18446744071562072064`, the following is effectively called:
```
recv(fd, buf, 18446744071562072064, 0);
```

This fails because that is beyond even our whole stack, and `recv()` is smart enough _on my system_ to not allow this; it fail with:

```
strerror(errno): Bad address
```

which, indeed means "The system detected an invalid pointer address in attempting to use a pointer argument of a call".

I have not investigated further whether it is possible to write an amount of memory which will satisfy the kernel's limitation here, and I don't know enough about other systems, and how they work.

In any case, I thought I would report this issue here instead of via a GH issue, because while it may be "unexploitable" to me, it may not be to others.
